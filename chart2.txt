    c_head1, c_head2, c_head3 = st.columns([6, 2, 2])
    with c_head1:
        st.markdown("#### 📊 Nifty 50 Price Chart")
    with c_head2:
        if st.button("🔄 Refresh Now", use_container_width=True):
            st.rerun()
    with c_head3:
        timeframe_choice = st.selectbox("Timeframe", ["5m (Intraday)", "15m (Short Term)", "1d (Daily)"], label_visibility="collapsed")
        tf_map = {"5m (Intraday)": ("5m", "5d"), "15m (Short Term)": ("15m", "1mo"), "1d (Daily)": ("1d", "6mo")}
        curr_int, curr_per = tf_map[timeframe_choice]

    @st.fragment(run_every="5s")
    def render_realtime_groww_chart(interval, period):
        _, spot_change_pct, candles = LiveDataFeed.fetch_nifty_spot_and_candles(interval=interval, period=period)
        
        # Groww Style Colors (Green if bullish, Red if bearish)
        line_color = "#00d09c" if spot_change_pct >= 0 else "#eb5b3c"
        fill_color = "rgba(0, 208, 156, 0.1)" if spot_change_pct >= 0 else "rgba(235, 91, 60, 0.1)"
        
        fig = go.Figure()
        
        # Main Area Chart (Groww style)
        fig.add_trace(go.Scatter(
            x=candles["time"],
            y=candles["Close"],
            mode="lines",
            name="Nifty",
            line=dict(color=line_color, width=2.5),
            fill='tozeroy',
            fillcolor=fill_color
        ))
        
        # 20 EMA (Subtle dotted line)
        fig.add_trace(go.Scatter(
            x=candles["time"],
            y=candles["EMA20"],
            mode="lines",
            name="20 EMA",
            line=dict(color="#58a6ff", width=1.5, dash="dot"),
        ))
        
        # Clean minimalist layout (Groww style)
        fig.update_layout(
            template="plotly_dark",
            paper_bgcolor="#0d1117",
            plot_bgcolor="#0d1117", # Darker background
            height=420,
            margin=dict(l=0, r=40, t=10, b=0), # Give room on right for yaxis
            xaxis_rangeslider_visible=False,
            hovermode="x unified",
            xaxis=dict(showgrid=False, zeroline=False, showticklabels=True, tickfont=dict(color="#8b949e")),
            yaxis=dict(showgrid=True, gridcolor="#161b22", zeroline=False, side="right", tickfont=dict(color="#8b949e")),
            showlegend=False
        )
        st.plotly_chart(fig, use_container_width=True)
        st.caption("💡 **Quick Rule**: When Nifty trades **above the blue dotted 20 EMA line**, CALL options have higher success probability. When below, PUT options are favored. *(Chart updates automatically every 5s)*")
        
    render_realtime_groww_chart(curr_int, curr_per)
